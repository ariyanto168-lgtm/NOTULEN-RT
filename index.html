Perbaikan fungsi Hapus (Masuk ke antrean jika sinyal jelek) function deleteRecord(tableName, id) { vibrate([50, 50]); let inputPin = prompt("⚠️ Otorisasi Hapus Data\nMasukkan PIN Admin:"); if(inputPin === null) return; if(inputPin !== getAdminPin()) { vibrate([50,50,50]); showToast("PIN Salah! Aksi Dibatalkan.", "❌"); return; } if(!window.confirm("Yakin ingin menghapus data ini secara permanen?")) return; APP_DATA[tableName] = APP_DATA[tableName].filter(x => x.id !== id); saveLocalData(); refreshAllUI(); showToast("Data dihapus", "🗑️"); let payload = { action: 'delete', table: tableName, id: id }; if(navigator.onLine && GAS_URL !== "") { sendToCloud(payload); } else { SYNC_QUEUE.push(payload); localStorage.setItem('rt_sync_queue', JSON.stringify(SYNC_QUEUE)); } } // 2. Perbaikan fungsi Reset function resetTableData(tableName, titleName) { vibrate([100, 100]); let inputPin = prompt(`⚠️ Otorisasi Reset Data\nMasukkan PIN Admin untuk mereset seluruh ${titleName}:`); if(inputPin === null) return; if(inputPin !== getAdminPin()) { vibrate([50,50,50]); showToast("PIN Salah! Aksi Dibatalkan.", "❌"); return; } if(!window.confirm(`PERINGATAN FINAL: Apakah Anda yakin mereset seluruh data ${titleName}? Tindakan ini akan menghapus data di server juga.`)) return; APP_DATA[tableName] = []; saveLocalData(); refreshAllUI(); showToast(`${titleName} direset`, "🧹"); let payload = { action: 'save_bulk', table: tableName, data: [] }; if(navigator.onLine && GAS_URL !== "") { sendToCloud(payload); } else { SYNC_QUEUE.push(payload); localStorage.setItem('rt_sync_queue', JSON.stringify(SYNC_QUEUE)); } } // 3. Perbaikan alur kirim Cloud (Cegah Sinkronisasi Bertumpuk) async function sendToCloud(payload, autoSync = true) { try { let res = await fetch(GAS_URL, { method: 'POST', body: JSON.stringify(payload) }); let result = await res.json(); // Hanya sync jika diminta dan berhasil if(result.status === "success" && autoSync) syncDataFromCloud(); } catch (e) { SYNC_QUEUE.push(payload); localStorage.setItem('rt_sync_queue', JSON.stringify(SYNC_QUEUE)); } } async function processQueue() { if (SYNC_QUEUE.length === 0 || !navigator.onLine || GAS_URL === "") return; let queueToProcess = [...SYNC_QUEUE]; SYNC_QUEUE = []; localStorage.setItem('rt_sync_queue', JSON.stringify(SYNC_QUEUE)); for (let item of queueToProcess) { // Kirim semua antrean ke server (tanpa menarik data dulu) await sendToCloud(item, false); } // Setelah semua antrean berhasil dikerjakan, BARU tarik data final dari server syncDataFromCloud(); } // 4. Perbaikan saat aplikasi pertama dibuka (Tunggu antrean lokal dikirim) window.onload = async () => { loadLocalData(); initDynamicYears(); let savedSec = localStorage.getItem('rt_active_section'); if(savedSec && document.getElementById(savedSec)) { navTo(savedSec, false); } else { navTo('sec-dashboard', false); } refreshAllUI(); loadKopSuratConfig(); loadPengaturanInput(); document.addEventListener('click', (e) => { let sRes = document.getElementById('search-results'); if(sRes && sRes.style.display === 'block' && !e.target.closest('.search-container')) { sRes.style.display = 'none'; } }); // LOGIKA PENCEGAH TABRAKAN DATA if (GAS_URL !== "") { if (SYNC_QUEUE.length > 0) { await processQueue(); // Tunggu antrean hapus terkirim } else { syncDataFromCloud(); // Tarik data jika tidak ada antrean } } else { showToast("Mode Lokal / Demo Aktif", "ℹ️"); initDemoData(); } window.addEventListener('online', async () => { document.getElementById('syncStatus').innerHTML = '🟢'; if (SYNC_QUEUE.length > 0) await processQueue(); else syncDataFromCloud(); }); window.addEventListener('offline', () => { document.getElementById('syncStatus').innerHTML = '🔴'; }); setInterval(() => { // Jangan auto-sync jika sedang ada data yang antre dikirim if(navigator.onLine && GAS_URL !== "" && SYNC_QUEUE.length === 0) { syncDataFromCloud(false); } }, 180000); };
